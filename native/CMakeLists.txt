cmake_minimum_required(VERSION 3.22)
project(dspfusion LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(JAMES_DIR "${CMAKE_CURRENT_LIST_DIR}/../app/src/main/cpp/libjamesdsp/Main/libjamesdsp/jni/jamesdsp")
file(GLOB_RECURSE JAMES_DSP CONFIGURE_DEPENDS "${JAMES_DIR}/jdsp/*.c")
file(GLOB_RECURSE VIPER_DSP CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/viper/src/*.cpp")
add_library(fusion_james OBJECT ${JAMES_DSP} "${JAMES_DIR}/cpthread.c" "${JAMES_DIR}/jamesdsp.c")
target_compile_definitions(fusion_james PRIVATE JAMESDSP_REFERENCE_IMPL DSP_FUSION_STREAMING AELI=JDSP_AELI)
target_include_directories(fusion_james PRIVATE "${JAMES_DIR}")
add_library(fusion_viper OBJECT ${VIPER_DSP})
target_compile_definitions(fusion_viper PRIVATE AELI=VIPER_AELI VERSION_CODE=20250107 VERSION_NAME="0.6.3-fusion")
target_include_directories(fusion_viper PRIVATE viper/src/include)
add_library(dspfusion SHARED fusion/FusionEffect.cpp tone/ToneProcessor.cpp peq/PeqProcessor.cpp surround/SurroundProcessor.cpp
    $<TARGET_OBJECTS:fusion_james> $<TARGET_OBJECTS:fusion_viper>)
target_include_directories(dspfusion PRIVATE viper/src/include)
foreach(target fusion_james fusion_viper dspfusion)
    target_compile_options(${target} PRIVATE -O2 -fvisibility=hidden -ffunction-sections -fdata-sections
        -include "${CMAKE_CURRENT_LIST_DIR}/fusion/compat.h")
endforeach()
if(ANDROID)
    target_link_libraries(dspfusion PRIVATE log m)
    target_link_options(dspfusion PRIVATE -Wl,--gc-sections -Wl,--exclude-libs,ALL
        "-Wl,--version-script=${CMAKE_CURRENT_LIST_DIR}/fusion/exports.map" -Wl,-z,max-page-size=16384)
else()
    foreach(target fusion_james fusion_viper dspfusion)
        target_include_directories(${target} PRIVATE tests/stubs)
    endforeach()
    target_link_libraries(dspfusion PRIVATE pthread m)
endif()
option(FUSION_BUILD_TESTS "Build native integration tests (host only)" OFF)
option(FUSION_SANITIZE "Enable address sanitizer for host integration tests" OFF)
if(FUSION_BUILD_TESTS AND NOT ANDROID)
    enable_testing()
    target_compile_definitions(fusion_james PRIVATE DSP_FUSION_TEST_ALLOCATIONS)
    add_executable(fusion_tests tests/FusionTests.cpp tests/StreamingTests.cpp)
    target_compile_definitions(fusion_tests PRIVATE DSP_FUSION_TEST_ALLOCATIONS)
    target_include_directories(fusion_tests PRIVATE viper/src/include)
    target_link_libraries(fusion_tests PRIVATE dspfusion)
    target_compile_options(fusion_tests PRIVATE -UNDEBUG -include "${CMAKE_CURRENT_LIST_DIR}/fusion/compat.h")
    add_test(NAME fusion_integration COMMAND fusion_tests)
    add_executable(peq_tests tests/PeqTests.cpp peq/PeqProcessor.cpp)
    target_compile_options(peq_tests PRIVATE -O2 -UNDEBUG)
    target_link_libraries(peq_tests PRIVATE pthread)
    add_test(NAME peq_audio COMMAND peq_tests)
    add_executable(surround_tests tests/SurroundTests.cpp surround/SurroundProcessor.cpp)
    target_compile_options(surround_tests PRIVATE -O2 -UNDEBUG)
    target_link_libraries(surround_tests PRIVATE pthread)
    add_test(NAME surround_audio COMMAND surround_tests)
    add_executable(bass_tests tests/BassTests.cpp tests/BassCoreTests.c
        $<TARGET_OBJECTS:fusion_james> $<TARGET_OBJECTS:fusion_viper>)
    target_include_directories(bass_tests PRIVATE "${JAMES_DIR}" viper/src/include tests/stubs)
    target_compile_options(bass_tests PRIVATE -O2 -UNDEBUG -include "${CMAKE_CURRENT_LIST_DIR}/fusion/compat.h")
    target_link_libraries(bass_tests PRIVATE pthread m)
    add_test(NAME bass_audio COMMAND bass_tests)
    if(FUSION_SANITIZE)
        foreach(target fusion_james fusion_viper dspfusion fusion_tests peq_tests surround_tests bass_tests)
            target_compile_options(${target} PRIVATE -fsanitize=address -fno-omit-frame-pointer)
        endforeach()
        target_link_options(dspfusion PRIVATE -fsanitize=address)
        target_link_options(fusion_tests PRIVATE -fsanitize=address)
        target_link_options(peq_tests PRIVATE -fsanitize=address)
        target_link_options(surround_tests PRIVATE -fsanitize=address)
        target_link_options(bass_tests PRIVATE -fsanitize=address)
    endif()
endif()
